   :param p1: End point of the segment.
   :type p1: array
   :param radius0: Radius at the start point.
   :type radius0: float
   :param radius1: Radius at the end point.
   :type radius1: float
   :param targetPt: Point at which to interpolate the radius.
   :type targetPt: array
